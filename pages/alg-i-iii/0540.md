$$

\bigotimes_{\lambda \in L} \Lambda(M_\lambda) & \xrightarrow{g_M} & \Lambda(M) \\

\bigotimes_{\lambda \in L} \Lambda(N_\lambda) & \xrightarrow{g_N} & \Lambda(N)
