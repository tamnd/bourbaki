$$
\begin{array}{ll}
\tilde{A}_1 & \\
\tilde{A}_l \quad (l \geq 2) & \text{(circuit à } l + 1 \text{ sommets)} \\
\tilde{B}_2 & \\
\tilde{B}_l \quad (l \geq 3) & (l + 1 \text{ sommets}) \\
\tilde{C}_l \quad (l \geq 3) & (l + 1 \text{ sommets}) \\
\tilde{D}_l \quad (l \geq 4) & (l + 1 \text{ sommets}) \\
\tilde{E}_6 & \\
\tilde{E}_7 & \\
\tilde{E}_8 & \\
\tilde{F}_4 & \\
\tilde{G}_2 &

$$
